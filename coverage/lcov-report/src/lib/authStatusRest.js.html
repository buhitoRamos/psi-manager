  const SUPABASE_URL = (process.env.REACT_APP_SUPABASE_URL || 'https://ljynujodigqqujjvyoud.supabase.co').replace(/\/+$/,'');

    Authorization: `Bearer ${SUPABASE_ANON_KEY}`,
    Accept: 'application/json',
    'Content-Type': 'application/json'
  };
